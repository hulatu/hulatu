{{- /* 返回文章列表缩略图信息，类型是 dict： src —— 缩略图 URL（本地方案是原图 220px 宽那一档） srcset —— 多尺寸候选串，给 用；拿不到就为空串 优先用 assets/img/thumbs/ 下的同名 webp：构建期按 64 / 128 / 220 三档输出， 64 给普通屏、128 给 2 倍屏、220 兜底给 3 倍屏。 找不到本地图就回退封面图，走 Cloudflare Images 取 64 / 128 两档。 没有封面时 src 为空串。 */ -}} {{- $src := "" -}} {{- $srcset := "" -}} {{- with .Params.cover -}} {{- $name := replace (path.Base .) "%20" "_" -}} {{- $stem := strings.TrimSuffix (path.Ext $name) $name -}} {{- $res := resources.Get (printf "img/thumbs/%s.webp" $stem) -}} {{- if $res -}} {{- $src = $res.RelPermalink -}} {{- $s64 := $res.Resize "64x64 webp" -}} {{- $s128 := $res.Resize "128x128 webp" -}} {{- $srcset = printf "%s 64w, %s 128w, %s %dw" $s64.RelPermalink $s128.RelPermalink $res.RelPermalink $res.Width -}} {{- else -}} {{- $src = partial "cf-image.html" (dict "src" . "params" "width=128,height=128,fit=cover,format=auto,quality=75,metadata=none") -}} {{- $s64 := partial "cf-image.html" (dict "src" . "params" "width=64,height=64,fit=cover,format=auto,quality=75,metadata=none") -}} {{- if ne $s64 $src -}} {{- $srcset = printf "%s 64w, %s 128w" $s64 $src -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "src" $src "srcset" $srcset) -}}
